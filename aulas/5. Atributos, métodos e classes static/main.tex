\section{Introdução}

\begin{frame}{O que significa \texttt{static}?}
    \begin{itemize}
        \item Em Java, o modificador \texttt{static} é usado para indicar que um membro (atributo, método ou classe aninhada) pertence à \textbf{classe} em si, e não a uma \textbf{instância} (objeto) específica.
        \item Isso significa que existe apenas uma cópia desse membro para todas as instâncias da classe.
        \item O membro estático é carregado na memória assim que a classe é carregada pela JVM.
    \end{itemize}
\end{frame}

\section{Atributos Estáticos}

\begin{frame}{Atributos Estáticos (Variáveis de Classe)}
    \begin{itemize}
        \item Também conhecidos como variáveis de classe.
        \item São declarados com a palavra-chave \texttt{static}.
        \item \textbf{Compartilhamento:} Todas as instâncias da classe compartilham a mesma variável. Se uma instância altera o valor, todas as outras veem a mudança.
        \item \textbf{Tempo de vida:} Existe durante toda a execução do programa (enquanto a classe estiver carregada).
    \end{itemize}
\end{frame}

\begin{frame}{Comparação: Instância vs. Estático}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{Atributos de Instância}
            \begin{itemize}
                \item Cada objeto tem sua própria cópia.
                \item Define o estado de um objeto individual.
                \item Criado quando o objeto é instanciado (\texttt{new}).
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{Atributos Estáticos}
            \begin{itemize}
                \item Uma única cópia para a classe toda.
                \item Define propriedades comuns a todos os objetos.
                \item Criado quando a classe é carregada.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}[fragile]{Exemplo de Código: Contador de Instâncias}
\begin{lstlisting}
public class Carro {
    public String modelo;
    public static int contadorProducao = 0; // Estatico

    public Carro(String modelo) {
        this.modelo = modelo;
        contadorProducao++; // Incrementa para todos
    }
}
\end{lstlisting}
    \begin{itemize}
        \item Toda vez que um \texttt{new Carro()} é chamado, o \texttt{contadorProducao} aumenta.
        \item Se criarmos 10 carros, o valor será 10 para todos.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Acessando Membros Estáticos}
    \begin{itemize}
        \item Embora seja possível acessar via instância, a boa prática é acessar via \textbf{Nome da Classe}.
    \end{itemize}
\begin{lstlisting}
Carro c1 = new Carro("Fusca");

// Forma recomendada (via classe)
System.out.println(Carro.contadorProducao);

// Forma nao recomendada (via instancia)
System.out.println(c1.contadorProducao); 
\end{lstlisting}
    \begin{alertblock}{Importante}
        O acesso via instância pode confundir outros programadores, dando a entender que a variável pertence apenas àquele objeto.
    \end{alertblock}
\end{frame}

\begin{frame}{Constantes (\texttt{static final})}
    \begin{itemize}
        \item A combinação de \texttt{static} e \texttt{final} é usada para criar constantes em Java.
        \item \texttt{static}: Compartilhada (economiza memória).
        \item \texttt{final}: O valor não pode ser alterado após a inicialização.
        \item Nomenclatura sugerida: \texttt{CAIXA\_ALTA} com underscores.
    \end{itemize}
    \begin{exampleblock}{Exemplo}
        \texttt{public static final double PI = 3.14159;}
    \end{exampleblock}
\end{frame}

\begin{frame}{Quando usar Atributos Estáticos?}
    \begin{enumerate}
        \item \textbf{Contadores:} Quando você precisa saber quantos objetos foram criados.
        \item \textbf{Constantes:} Valores fixos que nunca mudam (PI, Taxas fixas).
        \item \textbf{Configurações Compartilhadas:} Como um tema de cores global ou um caminho de log.
        \item \textbf{Economia de Memória:} Evita duplicar dados idênticos em milhares de objetos.
    \end{enumerate}
\end{frame}

\section{Métodos Estáticos}

\begin{frame}{O que são Métodos Estáticos?}
    \begin{itemize}
        \item São métodos que podem ser chamados sem a necessidade de instanciar um objeto da classe.
        \item Pertencem à classe, não a um objeto individual.
        \item Marcados com a palavra-chave \texttt{static}.
        \item \textbf{Uso principal:} Operações utilitárias que não dependem do estado interno de um objeto.
    \end{itemize}
\end{frame}

\begin{frame}{Características dos Métodos Estáticos}
    \begin{itemize}
        \item \textbf{Sem \texttt{this}:} Como não pertencem a uma instância, não existe a referência \texttt{this}.
        \item \textbf{Acesso limitado:} Só podem acessar diretamente outros membros estáticos da mesma classe.
        \item \textbf{Chamada:} Invocado usando \texttt{NomeDaClasse.nomeDoMetodo()}.
    \end{itemize}
    \begin{exampleblock}{Sintaxe}
        \texttt{public static int somar(int a, int b) \{ return a + b; \}}
    \end{exampleblock}
\end{frame}

\begin{frame}{Regras de Acesso}
    \centering
    \begin{tabular}{|l|c|c|}
        \hline
        \textbf{Acesso a partir de...} & \textbf{Membro Estático} & \textbf{Membro de Instância} \\ \hline
        Método Estático & Permitido \ding{51} & Proibido \ding{55} \\ \hline
        Método de Instância & Permitido \ding{51} & Permitido \ding{51} \\ \hline
    \end{tabular}
    \vspace{0.5cm}
    \begin{alertblock}{Por que o Método Estático não acessa a Instância?}
        Porque o método estático pode ser executado sem que nenhum objeto tenha sido criado no "Heap". Ele não saberia para qual objeto olhar.
    \end{alertblock}
\end{frame}

\begin{frame}[fragile]{Classes Utilitárias: \texttt{java.lang.Math}}
    \begin{itemize}
        \item Um dos maiores exemplos de uso de métodos estáticos no Java é a classe \texttt{Math}.
        \item Você não cria um "Objeto de Matemática", você apenas usa suas funções.
    \end{itemize}
\begin{lstlisting}
double raiz = Math.sqrt(25.0);
double potencia = Math.pow(2, 10);
int maior = Math.max(10, 20);
\end{lstlisting}
    \begin{itemize}
        \item Todos os métodos da classe \texttt{Math} são \texttt{static}.
    \end{itemize}
\end{frame}

\section{Blocos Estáticos}

\begin{frame}[fragile]{Blocos de Inicialização Estática}
    \begin{itemize}
        \item Usados para inicializar variáveis estáticas complexas.
        \item Executados apenas uma vez, quando a classe é carregada.
    \end{itemize}
\begin{lstlisting}
public class BancoDeDados {
    public static Connection conexao;

    static {
        System.out.println("Carregando driver...");
        // Logica complexa de inicializacao aqui
        conexao = conectar();
    }
}
\end{lstlisting}
\end{frame}

\begin{frame}{Ordem de Execução}
    \begin{enumerate}
        \item Membros Estáticos e Blocos Estáticos (na ordem que aparecem).
        \item Membros de Instância.
        \item Construtor.
    \end{enumerate}
    \begin{alertblock}{Importante}
        O bloco estático roda \textbf{antes} de qualquer objeto ser criado e apenas \textbf{uma vez} durante a vida da aplicação.
    \end{alertblock}
\end{frame}

\section{Classes Aninhadas Estáticas}

\begin{frame}{O que são Static Nested Classes?}
    \begin{itemize}
        \item Uma classe definida dentro de outra classe, marcada como \texttt{static}.
        \item Diferente das \textit{Inner Classes} (não-estáticas), a classe aninhada estática \textbf{não} possui uma referência para o objeto da classe externa.
        \item Comporta-se como uma classe de nível superior, mas está "escondida" dentro do namespace de outra classe.
    \end{itemize}
\end{frame}

\begin{frame}{Inner Class vs. Static Nested Class}
    \centering
    \begin{tabular}{|p{4.5cm}|p{4.5cm}|}
        \hline
        \textbf{Inner Class (Normal)} & \textbf{Static Nested Class} \\ \hline
        Precisa de uma instância da classe externa. & Não precisa de instância da classe externa. \\ \hline
        Pode acessar todos os membros (até privados) da externa. & Só pode acessar membros estáticos da classe externa. \\ \hline
        Declarada apenas como \texttt{class X}. & Declarada como \texttt{static class X}. \\ \hline
    \end{tabular}
\end{frame}

\begin{frame}[fragile]{Instanciação}
    \begin{itemize}
        \item Para criar uma instância de uma classe aninhada estática, usamos a sintaxe:
    \end{itemize}
\begin{lstlisting}
// Nao precisa de um objeto de 'Externa'
Externa.Aninhada objeto = new Externa.Aninhada();
\end{lstlisting}
    \begin{itemize}
        \item Comparação com Inner Class:
    \end{itemize}
\begin{lstlisting}
// Inner Class normal precisaria:
Externa ext = new Externa();
Externa.Interna in = ext.new Interna();
\end{lstlisting}
\end{frame}

\begin{frame}{Vantagens das Classes Estáticas}
    \begin{itemize}
        \item \textbf{Agrupamento Lógico:} Se uma classe é usada apenas por outra, faz sentido mantê-las juntas.
        \item \textbf{Encapsulamento:} A classe aninhada pode ser privada, escondendo detalhes de implementação do mundo exterior.
        \item \textbf{Legibilidade:} Facilita encontrar códigos relacionados.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Exemplo: Estrutura de Dados Helper}
\begin{lstlisting}
public class Lista {
    private static class No {
        int valor;
        No proximo;
        No(int v) { this.valor = v; }
    }

    private No cabeca;
    // ... metodos da lista ...
}
\end{lstlisting}
    \begin{itemize}
        \item O \texttt{No} não precisa saber qual \texttt{Lista} ele pertence individualmente, ele apenas armazena dados. Por isso ele é \texttt{static}.
    \end{itemize}
\end{frame}

\section{Exemplos e Padrões}

\begin{frame}[fragile]{Exemplo Completo: A Classe \texttt{Funcionario}}
\begin{lstlisting}
public class Funcionario {
    private String nome; 
    private double salario;
    private static int totalFuncionarios = 0; // Estatico

    public Funcionario(String nome, double salario) {
        this.nome = nome;
        this.salario = salario;
        totalFuncionarios++;
    }

    public static int getTotal() {
        return totalFuncionarios;
    }

    public void exibirDados() {
        System.out.println(nome + " - RS " + salario);
    }
}
\end{lstlisting}
\end{frame}

\begin{frame}{Boas Práticas: Quando NÃO usar \texttt{static}}
    \begin{itemize}
        \item \textbf{Estado Global:} Evite usar atributos estáticos para armazenar dados que mudam frequentemente e afetam todo o sistema (dificulta o rastreamento de bugs).
        \item \textbf{Testabilidade:} Membros estáticos são difíceis de "mockar" em testes unitários.
        \item \textbf{Polimorfismo:} Se você precisa que o comportamento mude via herança, não use métodos estáticos.
        \item \textbf{Abuso de 'Utils':} Não transforme Java em uma linguagem procedural criando apenas classes cheias de métodos estáticos.
    \end{itemize}
\end{frame}

\begin{frame}{Checklist de Aprendizado}
    \begin{itemize}
        \item[\ding{52}] \texttt{static} = Pertence à Classe.
        \item[\ding{52}] Atributos estáticos são compartilhados entre todos os objetos.
        \item[\ding{52}] Métodos estáticos não acessam \texttt{this} nem membros de instância.
        \item[\ding{52}] Blocos estáticos rodam uma única vez no carregamento.
        \item[\ding{52}] Classes aninhadas estáticas não precisam da classe externa instanciada.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Desafio Rápido}
    Analise o código abaixo. Ele compila?
\begin{lstlisting}
public class Teste {
    int x = 10;
    public static void main(String[] args) {
        System.out.println(x);
    }
}
\end{lstlisting}
    \medskip
    \begin{alertblock}{Resposta: Não!}
        O método \texttt{main} é \textbf{estático} e está tentando acessar a variável \texttt{x} que é de \textbf{instância} sem criar um objeto.
    \end{alertblock}
\end{frame}
