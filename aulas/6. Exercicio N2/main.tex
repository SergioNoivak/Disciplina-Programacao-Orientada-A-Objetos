\section{Introdução e Definição}

\begin{frame}{O que é a Compactação do Solo?}
    \begin{itemize}
        \item \textbf{Definição:} Processo de reorganização das partículas do solo que resulta no aumento da densidade e redução da porosidade.
        \item \textbf{Aspectos Físicos:}
        \begin{itemize}
            \item Expulsão do ar dos poros do solo.
            \item Redução do volume total do solo sob uma carga aplicada.
            \item Aumento da resistência mecânica à penetração.
        \end{itemize}
        \item \textbf{Estado Crítico:} Ocorre quando a pressão aplicada pelo maquinário ou animais excede a capacidade de suporte de carga do solo.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{1. Classe Abstrata: \texttt{MaquinaAgricola}}
\begin{lstlisting}[basicstyle=\tiny]
public abstract class MaquinaAgricola {
    protected double pesoToneladas;
    protected double pressaoSoloKPa;
    protected String tipoRodado;
    protected double resistenciaPenetracaoMPa;

    // Limiares cientificos (static final)
    public static final double RP_CRITICA_MIN = 2.0;
    public static final double RP_CRITICA_MAX = 2.5;
    protected static int totalMaquinas = 0;

    public MaquinaAgricola(double peso, double pressao, String rodado) {
        this.pesoToneladas = peso;
        this.pressaoSoloKPa = pressao;
        this.tipoRodado = rodado;
        totalMaquinas++;
    }

    public final boolean haRestricaoRadicular() {
        return resistenciaPenetracaoMPa >= RP_CRITICA_MIN;
    }

    public abstract String avaliarCompactacao();
}
\end{lstlisting}
\end{frame}

\begin{frame}{Conceitos Aplicados: \texttt{abstract}, \texttt{final} e \texttt{static}}
    \begin{itemize}
        \item \textbf{\texttt{abstract class}}: Define um modelo que não pode ser instanciado diretamente. Exige que cada máquina implemente sua própria lógica de compactação.
        \item \textbf{\texttt{static final}} (Constantes): \texttt{RP\_CRITICA\_MIN} é um valor global fixo baseado em literatura científica.
        \item \textbf{\texttt{static}} (Contador): \texttt{totalMaquinas} pertence à classe, permitindo rastrear quantas máquinas foram criadas no sistema.
        \item \textbf{\texttt{final method}}: \texttt{haRestricaoRadicular()} possui uma lógica universal que não deve ser alterada pelas subclasses.
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{3. Modelo de Produtividade: \texttt{static}}
\begin{lstlisting}[language=Java, basicstyle=\tiny]
public class ModeloProdutividade {
    public static final double A = 10.0;
    public static final double B = -4.0;
    public static final double C = -1.5;

    // Regressao quadratica envolvendo a maquina
    public static double calcularProdutividade(double densidade, MaquinaAgricola m) {
        double base = A + B * densidade + C * Math.pow(densidade, 2);
        // Penalidade se houver restricao radicular causada pela maquina
        return m.haRestricaoRadicular() ? base * 0.7 : base;
    }

    public static double produtividadePorRP(double rp) {
        if (rp < 2.0) return 100;
        if (rp < 2.5) return 80;
        return 50;
    }
}
\end{lstlisting}
\begin{itemize}
    \item \textbf{Métodos de Classe}: Não dependem de instâncias. Funcionam como uma "Calculadora Científica" de produtividade disponível para todo o sistema.
\end{itemize}
\end{frame}
